<script>
    // 1. Data Structure: Edit images and prices here
    const products = [
        { id: 1, name: "Premium Olive Oil Soap", price: 25, img: "https://images.unsplash.com/photo-1600857062241-98e5dba7f214?auto=format&fit=crop&w=500" },
        { id: 2, name: "Luxury Silk Scarf", price: 85, img: "https://images.unsplash.com/photo-1584917865442-de89df76afd3?auto=format&fit=crop&w=500" },
        { id: 3, name: "Organic Face Serum", price: 45, img: "https://images.unsplash.com/photo-1620916566398-39f1143ab7be?auto=format&fit=crop&w=500" },
        { id: 4, name: "Handmade Leather Bag", price: 120, img: "https://images.unsplash.com/photo-1548036328-c9fa89d128fa?auto=format&fit=crop&w=500" }
    ];

    let cart = [];

    // 2. Render Products
    const productsContainer = document.getElementById('products');
    products.forEach(p => {
        productsContainer.innerHTML += `
            <div class="product-card">
                <img src="${p.img}" class="product-img" alt="${p.name}">
                <div class="product-info">
                    <h3>${p.name}</h3>
                    <p class="product-price">$${p.price}</p>
                    <button class="add-btn" onclick="addToCart(${p.id})">Add to Cart</button>
                </div>
            </div>
        `;
    });

    // 3. Cart Logic
    function addToCart(id) {
        const product = products.find(p => p.id === id);
        cart.push(product);
        updateCartUI();
        alert(product.name + " added to cart!");
    }

    function updateCartUI() {
        document.getElementById('cart-count').innerText = cart.length;
    }

    function toggleModal() {
        const modal = document.getElementById('cart-modal');
        modal.style.display = modal.style.display === 'block' ? 'none' : 'block';
        
        const list = document.getElementById('cart-items-list');
        list.innerHTML = cart.map(item => `<p>• ${item.name} - $${item.price}</p>`).join('');
    }

    // 4. Form Handling (Order to Email/WhatsApp)
    document.getElementById('order-form').addEventListener('submit', function(e) {
        e.preventDefault();
        const name = document.getElementById('name').value;
        const phone = document.getElementById('phone').value;
        const address = document.getElementById('address').value;
        const items = cart.map(i => i.name).join(', ');

        // Using WhatsApp API for order delivery
        const message = `New Order from ${name}\nItems: ${items}\nPhone: ${phone}\nAddress: ${address}`;
        const whatsappUrl = `https://wa.me/216XXXXXXXX?text=${encodeURIComponent(message)}`; // Replace with your Tunisian Number
        
        window.open(whatsappUrl, '_blank');
        cart = [];
        updateCartUI();
        toggleModal();
    });
</script>
